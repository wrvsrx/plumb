==================
Parsed inline elements
==================

Text with `*[emphasis] and `outer[before `inner[nested] after].

---

(document
  (blank_line)
  (paragraph
    (inline_content
      (text)
      (inline_element
        (introducer)
        (inline_kind)
        (parsed_inline_content
          (inline_text)))
      (text)
      (inline_element
        (introducer)
        (inline_kind)
        (parsed_inline_content
          (inline_text)
          (inline_element
            (introducer)
            (inline_kind)
            (parsed_inline_content
              (inline_text)))
          (inline_text)))
      (text))))
